from django.contrib import admin
from .models import OrchidSpecies, CultivationCondition

@admin.register(OrchidSpecies)
class OrchidSpeciesAdmin(admin.ModelAdmin):
    list_display = ('name', 'origin_zone', 'temperature_min', 'temperature_max')
    search_fields = ('name', 'origin_zone')

@admin.register(CultivationCondition)
class CultivationConditionAdmin(admin.ModelAdmin):
    list_display = ('orchid', 'location_type', 'current_temperature', 'last_watered')
    list_filter = ('location_type',)
    search_fields = ('orchid__name',)
